### 1. Ajout des filtres dans TaskProvider (25 min)

Ajoutez un `enum` pour les filtres et les getters correspondants dans `task_provider.dart` :

```dart
enum TaskFilter { all, active, completed }
```

```dart
class TaskProvider extends ChangeNotifier {
  final List<Task> _tasks = [];
  TaskFilter _currentFilter = TaskFilter.all;

  TaskFilter get currentFilter => _currentFilter;

  void setFilter(TaskFilter filter) {
    _currentFilter = filter;
    notifyListeners();
  }

  List<Task> get tasks {
    switch (_currentFilter) {
      case TaskFilter.all:
        return List.unmodifiable(_tasks);
      case TaskFilter.active:
        return List.unmodifiable(_tasks.where((t) => !t.isDone));
      case TaskFilter.completed:
        return List.unmodifiable(_tasks.where((t) => t.isDone));
    }
  }

  // Toujours sur la liste complète, indépendamment du filtre
  int get totalCount     => _tasks.length;
  int get pendingCount   => _tasks.where((t) => !t.isDone).length;
  int get completedCount => _tasks.where((t) => t.isDone).length;
  double get completionRate =>
      _tasks.isEmpty ? 0 : completedCount / _tasks.length;

  // ... reste du code inchangé
}
```

### 2. Navigation par onglets avec TabBar (35 min)

Dans `HomeScreen`, transformez la structure pour utiliser un `DefaultTabController` :

```dart
class HomeScreen extends StatelessWidget {
  const HomeScreen({super.key});

  @override
  Widget build(BuildContext context) {
    final provider = context.watch<TaskProvider>();

    return DefaultTabController(
      length: 3,
      child: Scaffold(
        appBar: AppBar(
          title: Text('Mes tâches (${provider.pendingCount} restantes)'),
          bottom: const TabBar(
            tabs: [
              Tab(text: 'Toutes'),
              Tab(text: 'En cours'),
              Tab(text: 'Terminées'),
            ],
          ),
        ),
        body: TabBarView(
          children: [
            _TaskList(filter: TaskFilter.all),
            _TaskList(filter: TaskFilter.active),
            _TaskList(filter: TaskFilter.completed),
          ],
        ),
        floatingActionButton: FloatingActionButton(
          onPressed: () => Navigator.pushNamed(context, '/add'),
          child: const Icon(Icons.add),
        ),
      ),
    );
  }
}
```

Créez un widget privé `_TaskList` qui affiche la liste filtrée :

```dart
class _TaskList extends StatelessWidget {
  final TaskFilter filter;
  const _TaskList({required this.filter});

  @override
  Widget build(BuildContext context) {
    // Appliquer le filtre avant de lire
    WidgetsBinding.instance.addPostFrameCallback((_) {
      context.read<TaskProvider>().setFilter(filter);
    });

    final tasks = context.watch<TaskProvider>().tasks;

    if (tasks.isEmpty) return const EmptyState();

    return ListView.builder(
      itemCount: tasks.length,
      itemBuilder: (context, index) {
        final task = tasks[index];
        return Dismissible(
          key: Key(task.id),
          direction: DismissDirection.endToStart,
          background: Container(
            color: Colors.red,
            alignment: Alignment.centerRight,
            padding: const EdgeInsets.only(right: 16),
            child: const Icon(Icons.delete, color: Colors.white),
          ),
          onDismissed: (_) => context.read<TaskProvider>().deleteTask(task.id),
          child: TaskTile(task: task),
        );
      },
    );
  }
}
```

> **Alternative :** Si la gestion du filtre dans `_TaskList` vous semble complexe, vous pouvez passer directement la liste filtrée en paramètre depuis `HomeScreen`. Choisissez l'approche que vous comprenez le mieux.

### 3. Synchronisation du filtre avec l'onglet actif (20 min)

Une approche plus propre consiste à écouter les changements d'onglet via un `TabController` explicite :

```dart
class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});
  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> with SingleTickerProviderStateMixin {
  late TabController _tabController;

  static const _filters = [TaskFilter.all, TaskFilter.active, TaskFilter.completed];

  @override
  void initState() {
    super.initState();
    _tabController = TabController(length: 3, vsync: this);
    _tabController.addListener(() {
      if (!_tabController.indexIsChanging) {
        context.read<TaskProvider>().setFilter(_filters[_tabController.index]);
      }
    });
  }

  @override
  void dispose() {
    _tabController.dispose();
    super.dispose();
  }

  // ...
}
```
